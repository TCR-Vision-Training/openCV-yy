#include <opencv2/opencv.hpp>   // OpenCV 核心库，包含图像处理、形态学、轮廓检测等功能
#include <iostream>             // 标准输入输出流，用于 cout 打印调试信息
#include <vector>               // STL 动态数组，用于存储轮廓、缺陷列表
#include <string>               // 字符串处理，用于缺陷编号如 "D1"
#include <algorithm>            // 标准算法库，用于 sort 排序
#include <iomanip>              // 格式化输出（如 fixed、setprecision）
#include <sstream>              // 字符串流，用于拼接编号和统计信息
#include <cmath>                // 数学函数库，提供 sqrt、pow 等

using namespace cv;             // 使用 OpenCV 命名空间，避免每次写 cv::
using namespace std;            // 使用标准命名空间

// 缺陷类型枚举：划痕（细长线条）、暗斑（深色圆斑）、亮点（浅色凸起）
enum DefectType { SCRATCH, DARK_SPOT, BRIGHT_SPOT };

// 缺陷结构体：存储每个缺陷的外接矩形、类型、编号标签
struct Defect {
    Rect bbox;                  // 外接矩形，用于画框和定位
    DefectType type;            // 缺陷类别（划痕/暗斑/亮点）
    string label;               // 显示标签，如 "D1"、"D2"
};

// 计算两个矩形中心点之间的欧氏距离
// 用于后续判断两个检测框是否属于同一个缺陷，进行合并去重
double centerDist(const Rect& a, const Rect& b)
{
    Point ca(a.x + a.width / 2, a.y + a.height / 2);  // 矩形 a 的中心点
    Point cb(b.x + b.width / 2, b.y + b.height / 2);  // 矩形 b 的中心点
    return sqrt(pow(ca.x - cb.x, 2) + pow(ca.y - cb.y, 2));  // 欧氏距离公式
}

// 非极大值抑制（NMS）：合并距离相近的同类缺陷
// 解决霍夫检测或阈值分割后，一个真实缺陷产生多个重叠框的问题
vector<Defect> mergeNearbyDefects(const vector<Defect>& defects)
{
    vector<Defect> merged;                      // 存储合并后的结果
    vector<bool> used(defects.size(), false);   // 标记数组，防止重复处理

    for (size_t i = 0; i < defects.size(); i++)
    {
        if (used[i]) continue;                  // 已合并过的跳过

        Defect curr = defects[i];               // 当前缺陷作为合并基准
        used[i] = true;

        // 遍历后续缺陷，寻找同类型且距离近的邻居进行合并
        for (size_t j = i + 1; j < defects.size(); j++)
        {
            if (used[j] || defects[j].type != curr.type) continue;  // 不同类型不合并

            double dist = centerDist(curr.bbox, defects[j].bbox);
            // 划痕通常较长，合并阈值放宽到 150px；斑点较小，阈值 60px
            double thresh = (curr.type == SCRATCH) ? 150.0 : 60.0;

            if (dist < thresh)
            {
                // 扩展外接矩形，将两个框合并成一个大框
                int x1 = min(curr.bbox.x, defects[j].bbox.x);
                int y1 = min(curr.bbox.y, defects[j].bbox.y);
                int x2 = max(curr.bbox.x + curr.bbox.width,  defects[j].bbox.x + defects[j].bbox.width);
                int y2 = max(curr.bbox.y + curr.bbox.height, defects[j].bbox.y + defects[j].bbox.height);
                curr.bbox = Rect(x1, y1, x2 - x1, y2 - y1);
                used[j] = true;  // 标记为已合并
            }
        }
        merged.push_back(curr);
    }
    return merged;
}

// 计算轮廓区域内的平均灰度值
// 辅助判断：暗斑的灰度值应明显低于背景，可作为二次验证依据
double meanGrayInContour(const Mat& grayImg, const vector<Point>& contour)
{
    Mat mask = Mat::zeros(grayImg.size(), CV_8UC1);  // 创建全黑掩膜
    // 在掩膜上绘制填充轮廓（白色），后续只对白色区域计算均值
    drawContours(mask, vector<vector<Point>>{contour}, -1, Scalar(255), -1);
    return mean(grayImg, mask)[0];  // 返回轮廓内的平均灰度（0~255）
}

int main()
{
    // 读取待检测的金属表面图像
    Mat img = imread("../question3/picture3/picture3.png");
    if (img.empty())
    {
        cout << "图片读取失败" << endl;
        return -1;
    }

    // 转换为灰度图，后续处理基于单通道亮度信息
    Mat gray;
    cvtColor(img, gray, COLOR_BGR2GRAY);

    // ========== 1. 背景估计 ==========
    // 使用大核高斯模糊（51×51）平滑图像，消除细小缺陷，保留光照渐变背景
    // 这是"背景减除法"的核心：大核模糊 = 低通滤波 = 背景近似
    Mat bg;
    GaussianBlur(gray, bg, Size(51, 51), 0);

    // ========== 2. 背景减除 ==========
    // 金属表面有光照渐变，不能直接用全局阈值
    // 背景减除后，缺陷与背景的差异被放大，不受渐变影响
    Mat dark_diff, bright_diff;
    subtract(bg, gray, dark_diff);   // 背景亮、缺陷暗 → 暗缺陷（划痕+暗斑）
    subtract(gray, bg, bright_diff); // 缺陷亮、背景暗 → 亮缺陷（亮点）

    // ========== 3. 阈值分割 ==========
    // 对差分图像进行二值化，分离出暗/亮缺陷区域
    Mat dark_mask, bright_mask;
    threshold(dark_diff, dark_mask, 10, 255, THRESH_BINARY);    // 暗缺陷阈值较低，保留底部细短线
    threshold(bright_diff, bright_mask, 15, 255, THRESH_BINARY); // 亮缺陷阈值，检测中间白色亮点

    // ========== 4. 形态学处理 ==========
    // 暗缺陷：闭运算连接断裂的划痕线条；开运算去除孤立噪点
    // 注意：闭运算核不能太大，否则会粘连相邻的不同缺陷（如右下暗点与划痕）
    Mat kernel_close = getStructuringElement(MORPH_RECT, Size(5, 5));
    morphologyEx(dark_mask, dark_mask, MORPH_CLOSE, kernel_close);

    Mat kernel_open = getStructuringElement(MORPH_ELLIPSE, Size(3, 3));
    morphologyEx(dark_mask, dark_mask, MORPH_OPEN, kernel_open);

    // 亮缺陷：同样进行开闭运算去噪
    morphologyEx(bright_mask, bright_mask, MORPH_OPEN, kernel_open);
    morphologyEx(bright_mask, bright_mask, MORPH_CLOSE, kernel_close);

    // ========== 5. 检测暗缺陷 ==========
    // 提取暗缺陷掩膜中的所有外轮廓
    vector<vector<Point>> contours;
    findContours(dark_mask, contours, RETR_EXTERNAL, CHAIN_APPROX_SIMPLE);

    vector<Defect> raw_defects;  // 原始缺陷列表（未合并）

    for (size_t i = 0; i < contours.size(); i++)
    {
        double area = contourArea(contours[i]);
        if (area < 15) continue;  // 面积过滤：去除微小噪点，但保留底部细短线（面积较小）

        Rect bbox = boundingRect(contours[i]);  // 轮廓的外接正矩形

        // 计算最小外接旋转矩形，用于获取真实长宽比
        // boundingRect 对倾斜物体不敏感，minAreaRect 更精确
        RotatedRect minRect = minAreaRect(contours[i]);
        float w = minRect.size.width;
        float h = minRect.size.height;
        if (w < h) swap(w, h);  // 保证 w 是长边，h 是短边
        float aspectRatio = w / max(h, 1.0f);  // 长宽比：划痕很大，暗斑接近1

        // 计算轮廓周长和圆形度
        // 圆形度 = 4π×面积 / 周长²，正圆为1，越不规则越小
        double perimeter = arcLength(contours[i], true);
        double circularity = (perimeter > 0) ? (4 * CV_PI * area / (perimeter * perimeter)) : 0;

        // 计算轮廓内平均灰度，作为暗斑的辅助验证（暗斑灰度应显著偏低）
        double meanGray = meanGrayInContour(gray, contours[i]);

        // 调试输出：运行时可观察各轮廓特征，用于调参
       // cout << "Dark contour " << i << ": area=" << area
        //     << ", aspect=" << aspectRatio
        //     << ", circ=" << circularity
         //    << ", meanGray=" << meanGray << endl;

        // 分类逻辑：
        // 划痕：长宽比大（>3.5）、细长、面积适中
        if (aspectRatio > 3.5 && area < 20000)
        {
            raw_defects.push_back({bbox, SCRATCH, ""});
        }
        // 暗斑：面积适中（300~600）、接近正圆（圆度>0.8）、长宽比接近1
        else if (area >= 300 && area <= 600 && circularity > 0.80 && aspectRatio < 1.5)
        {
            raw_defects.push_back({bbox, DARK_SPOT, ""});
        }
    }

    // ========== 6. 检测亮缺陷 ==========
    // 重新提取亮缺陷掩膜的轮廓
    contours.clear();
    findContours(bright_mask, contours, RETR_EXTERNAL, CHAIN_APPROX_SIMPLE);

    for (size_t i = 0; i < contours.size(); i++)
    {
        double area = contourArea(contours[i]);
        // 亮点面积通常较小且集中，过滤光晕产生的大面积虚假区域
        if (area > 150 && area < 400)
        {
            Rect bbox = boundingRect(contours[i]);
            raw_defects.push_back({bbox, BRIGHT_SPOT, ""});
        }
    }

    // ========== 7. 合并重复检测 ==========
    // 对原始缺陷列表进行 NMS，合并距离过近的同类缺陷
    vector<Defect> defects = mergeNearbyDefects(raw_defects);

    // ========== 8. 排序并编号 ==========
    // 按从上到下、从左到右的顺序排列，使编号 D1、D2... 与视觉顺序一致
    sort(defects.begin(), defects.end(), [](const Defect& a, const Defect& b)
    {
        if (abs(a.bbox.y - b.bbox.y) > 40)   // y 方向差距大时，按上下排
            return a.bbox.y < b.bbox.y;
        return a.bbox.x < b.bbox.x;           // y 接近时，按左右排
    });

    // 生成标签 D1、D2、D3...
    for (size_t i = 0; i < defects.size(); i++)
    {
        ostringstream oss;
        oss << "D" << (i + 1);
        defects[i].label = oss.str();
    }

    // ========== 9. 绘制结果 ==========
    int scratch_cnt = 0, dark_cnt = 0, bright_cnt = 0;

    for (const auto& d : defects)
    {
        // 用红色矩形框标出缺陷位置
        rectangle(img, d.bbox, Scalar(0, 0, 255), 2);

        // 标签放在矩形框左上角上方；若太靠顶边，则移到框内显示
        int label_y = d.bbox.y - 5;
        if (label_y < 20) label_y = d.bbox.y + 25;

        putText(img, d.label, Point(d.bbox.x, label_y),
                FONT_HERSHEY_SIMPLEX, 0.7, Scalar(0, 0, 255), 2);

        // 分类计数
        if (d.type == SCRATCH) scratch_cnt++;
        else if (d.type == DARK_SPOT) dark_cnt++;
        else if (d.type == BRIGHT_SPOT) bright_cnt++;
    }

    // ========== 绘制左上角统计信息 ==========
    ostringstream info;
    info << "Total Defects: " << defects.size();
    // 在图像左上角显示总缺陷数，红色字体
    putText(img, info.str(), Point(20, 35),
            FONT_HERSHEY_SIMPLEX, 1.0, Scalar(0, 0, 255), 2);

    // ========== 10. 输出并保存 ==========
    cout << "--------------------------------" << endl;
    cout << "划痕(Scratch): " << scratch_cnt << endl;
    cout << "暗斑(DarkSpot): " << dark_cnt << endl;
    cout << "亮点(BrightSpot): " << bright_cnt << endl;
    cout << "总缺陷数: " << defects.size() << endl;
    cout << "--------------------------------" << endl;

    // 保存标注结果到文件
    imwrite("../question3/result3.png", img);
    // 弹出窗口显示结果
    imshow("Defect Detection", img);
    waitKey(0);

return 0;
}
