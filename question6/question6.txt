#include <opencv2/opencv.hpp>
#include <iostream>
#include <vector>

using namespace cv;
using namespace std;


int main()
{

    Mat img =
    imread("../question6/picture6/picture6.png");


    if(img.empty())
    {
        cout<<"图片读取失败"<<endl;
        return -1;
    }



    //========================
    // 1. HSV提取蓝色点
    //========================

    Mat hsv;

    cvtColor(
        img,
        hsv,
        COLOR_BGR2HSV
    );


    Mat blueMask;


    // 蓝色范围
    inRange(
        hsv,
        Scalar(100,80,50),
        Scalar(140,255,255),
        blueMask
    );



    //========================
    // 2. 找蓝点中心
    //========================

    vector<vector<Point>> blueContours;


    findContours(
        blueMask,
        blueContours,
        RETR_EXTERNAL,
        CHAIN_APPROX_SIMPLE
    );


    vector<Point> centers;


    for(auto c:blueContours)
    {

        double area =
            contourArea(c);


        //过滤噪声
        if(area<5)
            continue;



        Moments m=moments(c);


        Point p(
            m.m10/m.m00,
            m.m01/m.m00
        );


        centers.push_back(p);

    }



    cout<<"检测蓝点数量:"
        <<centers.size()
        <<endl;



    //========================
    // 3. 建立marker
    // 一个蓝点代表一个颗粒
    //========================


    Mat markers =
        Mat::zeros(
            img.size(),
            CV_32S
        );



    for(int i=0;i<centers.size();i++)
    {

        circle(
            markers,
            centers[i],
            8,
            Scalar(i+1),
            -1
        );

    }



    //========================
    // 4. watershed分割
    //========================


    watershed(
        img,
        markers
    );



    //========================
    // 5. 根据marker画边界
    //========================


    int count=0;


    for(int i=1;i<=centers.size();i++)
    {


        Mat mask =
            markers==i;


        vector<vector<Point>> contours;


        findContours(
            mask,
            contours,
            RETR_EXTERNAL,
            CHAIN_APPROX_SIMPLE
        );


        if(contours.empty())
            continue;



        double area =
            contourArea(contours[0]);


        if(area<100)
            continue;



        count++;



        //绿色细边界

        drawContours(
            img,
            contours,
            0,
            Scalar(0,255,0),
            1
        );



        //中心蓝点

        circle(
            img,
            centers[i-1],
            3,
            Scalar(255,0,0),
            -1
        );



        putText(
            img,
            to_string(count),
            centers[i-1]+Point(-10,5),
            FONT_HERSHEY_SIMPLEX,
            0.6,
            Scalar(255,0,0),
            2
        );


    }



    //========================
    // 6. 左上角数量
    //========================


    putText(
        img,
        "Valid Particles: "
        +to_string(count),
        Point(30,40),
        FONT_HERSHEY_SIMPLEX,
        1,
        Scalar(0,0,255),
        2
    );



    cout<<"最终颗粒数量:"
        <<count
        <<endl;



    imwrite(
        "../question6/result6.png",
        img
    );


    imshow(
        "Particle Count",
        img
    );


    waitKey(0);


    return 0;
}
