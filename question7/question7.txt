#include <opencv2/opencv.hpp>
#include <opencv2/ximgproc.hpp>

#include <iostream>
#include <vector>

using namespace cv;
using namespace std;


//================================
//计算骨架长度
//================================

double calcLength(Mat skeleton)
{
    double length=0;


    int dx[8]={1,1,0,-1,-1,-1,0,1};
    int dy[8]={0,1,1,1,0,-1,-1,-1};


    for(int y=1;y<skeleton.rows-1;y++)
    {
        for(int x=1;x<skeleton.cols-1;x++)
        {

            if(skeleton.at<uchar>(y,x)>0)
            {

                for(int k=0;k<8;k++)
                {

                    int nx=x+dx[k];
                    int ny=y+dy[k];


                    if(skeleton.at<uchar>(ny,nx)>0)
                    {

                        if(k%2==0)
                            length+=1.0;
                        else
                            length+=1.414;

                    }

                }

            }

        }
    }


    return length/2.0;
}






int main()
{


    //==========================
    //读取图片
    //==========================

    Mat img =
    imread("../question7/picture7/picture7.png");


    if(img.empty())
    {
        cout<<"图片读取失败"<<endl;
        return -1;
    }



    Mat gray;


    cvtColor(
            img,
            gray,
            COLOR_BGR2GRAY
    );





    //==========================
    //1 黑帽增强暗裂纹
    //==========================

    Mat blackhat;


    Mat bhKernel =
    getStructuringElement(
            MORPH_RECT,
            Size(25,25)
    );



    morphologyEx(
            gray,
            blackhat,
            MORPH_BLACKHAT,
            bhKernel
    );





    //==========================
    //2 二值化
    //==========================

    Mat binary;


    threshold(
            blackhat,
            binary,
            10,
            255,
            THRESH_BINARY
    );





    //==========================
    //3 小范围连接裂纹
    //==========================


    Mat kernel =
    getStructuringElement(
            MORPH_ELLIPSE,
            Size(5,5)
    );



    morphologyEx(
            binary,
            binary,
            MORPH_CLOSE,
            kernel
    );





    //==========================
    //4 轻微连接断点
    //==========================


    Mat connect;


    Mat connectKernel =
    getStructuringElement(
            MORPH_ELLIPSE,
            Size(9,9)
    );



    morphologyEx(
            binary,
            connect,
            MORPH_CLOSE,
            connectKernel
    );



    dilate(
            connect,
            connect,
            getStructuringElement(
                    MORPH_RECT,
                    Size(3,3)
            )
    );





    imshow(
            "connect",
            connect
    );





    //==========================
    //5 找裂纹区域
    //==========================


    vector<vector<Point>> contours;



    findContours(
            connect,
            contours,
            RETR_EXTERNAL,
            CHAIN_APPROX_NONE
    );




    Mat result =
    img.clone();



    int crackCount=0;


    double maxLength=0;




    cout<<"===================="<<endl;





    for(int i=0;i<contours.size();i++)
    {


        double area =
        contourArea(contours[i]);



        //过滤噪声和焊缝

        if(area<30 || area>3000)
            continue;



        crackCount++;





        //=====================
        //生成单条mask
        //=====================


        Mat mask =
        Mat::zeros(
                gray.size(),
                CV_8UC1
        );



        drawContours(
                mask,
                contours,
                i,
                Scalar(255),
                FILLED
        );






        //=====================
        //骨架提取
        //=====================


        Mat skeleton;



        ximgproc::thinning(
                mask,
                skeleton,
                ximgproc::THINNING_ZHANGSUEN
        );




        double len =
        calcLength(skeleton);



        cout
        <<"Crack "
        <<crackCount
        <<" length="
        <<len
        <<" pixels"
        <<endl;




        if(len>maxLength)
            maxLength=len;






        //=====================
        //红色绘制骨架
        //=====================


        for(int y=0;y<skeleton.rows;y++)
        {
            for(int x=0;x<skeleton.cols;x++)
            {

                if(skeleton.at<uchar>(y,x)>0)
                {

                    result.at<Vec3b>(y,x)
                    =
                    Vec3b(0,0,255);

                }

            }
        }





        //编号

        Rect box =
        boundingRect(contours[i]);



        putText(
                result,
                "Crack-"+to_string(crackCount),
                Point(
                        box.x,
                        box.y-10
                ),
                FONT_HERSHEY_SIMPLEX,
                0.8,
                Scalar(0,255,0),
                2
        );

    }






    cout<<"===================="<<endl;


    cout
    <<"裂纹数量:"
    <<crackCount
    <<endl;


    cout
    <<"最长裂纹:"
    <<maxLength
    <<" pixels"
    <<endl;







    //==========================
    //显示数量
    //==========================


    putText(
            result,
            "Cracks Detected: "
            +to_string(crackCount),
            Point(30,50),
            FONT_HERSHEY_SIMPLEX,
            1.2,
            Scalar(0,255,0),
            3
    );





    imwrite(
            "result.png",
            result
    );



    imshow(
            "Crack Result",
            result
    );



    waitKey(0);


    return 0;
}
